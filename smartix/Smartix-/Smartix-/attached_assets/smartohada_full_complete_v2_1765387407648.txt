1) High-quality selection logic (backend helper)
File: backend/app/utils_video_quality.py

Purpose: from a list of available video files (from API or scraped page), pick the best quality stream (prefer 4K -> 1080p -> 720p -> fallback to largest file size)

Code (place in backend/app/utils_video_quality.py):

```python
def pick_best_video(candidate_files):
    """candidate_files: list of dicts with keys: url, width, height, quality_label, filesize (optional)
    Returns chosen file url"""
    if not candidate_files:
        return None
    # normalize width/height if provided, compute score
    def score(f):
        w = f.get('width') or 0
        h = f.get('height') or 0
        size = f.get('filesize') or 0
        # prefer resolution then filesize
        return (w*h, size)
    best = max(candidate_files, key=score)
    return best.get('url')
```

Usage: Each fetcher should extract `video_files` or sources list and call `pick_best_video` to set `video_url` field in the normalized doc.

```python
import requests
from bs4 import BeautifulSoup
from app.utils import make_video_doc

def fetch_videvo(per_page=12):
    items = []
    try:
        r = requests.get('https://www.videvo.net/stock-video-footage/', timeout=10)
        if r.status_code!=200: return items
        soup = BeautifulSoup(r.text, 'html.parser')
        cards = soup.select('.media__link')[:per_page]
        for c in cards:
            href = c.get('href')
            if not href: continue
            page = requests.get('https://www.videvo.net'+href, timeout=8)
            if page.status_code!=200: continue
            sp = BeautifulSoup(page.text,'html.parser')
            dl = sp.find('a', {'class':'download-button'}) or sp.find('a', href=True)
            video_url = None
            if dl and dl.get('href'):
                video_url = dl.get('href')
            thumb = sp.find('meta', {'property':'og:image'})
            thumb = thumb.get('content') if thumb else None
            title = sp.find('meta', {'property':'og:title'})
            title = title.get('content') if title else 'Videvo clip'
            source_id = href.strip('/').split('/')[-1]
            if video_url:
                items.append(make_video_doc('videvo', source_id, video_url, thumb, title, None, tags=[]))
    except Exception:
        pass
    return items
```

```python
def fetch_archive_org(per_page=12):

        if r.status_code!=200: return items
        soup = BeautifulSoup(r.text,'html.parser')
        cards = soup.select('.item-ia')[:per_page]
        for c in cards:
            link = c.find('a', href=True)
            href = link['href'] if link else None
            if not href: continue
            page = requests.get('https://archive.org'+href, timeout=8)

            if not sources: continue
            video_url = sources[0].get('href')
            if video_url.startswith('/'): video_url = 'https://archive.org' + video_url
            thumb = sp.find('meta', {'property':'og:image'})
            thumb = thumb.get('content') if thumb else None
            title = sp.find('meta', {'property':'og:title'})
            title = title.get('content') if title else 'Archive clip'
            source_id = href.strip('/').split('/')[-1]
            items.append(make_video_doc('archive', source_id, video_url, thumb, title, None, tags=[]))
    except Exception:
        pass
    return items
```
