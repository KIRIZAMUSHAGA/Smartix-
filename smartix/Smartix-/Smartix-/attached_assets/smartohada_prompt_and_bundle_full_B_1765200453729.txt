from pydantic import BaseModel, Field
from typing import Optional
from datetime import datetime
from bson import ObjectId

class PyObjectId(ObjectId):
    @classmethod
    def __get_validators__(cls):
        yield cls.validate
    @classmethod
    def validate(cls, v):
        if not ObjectId.is_valid(v):
            raise ValueError("Invalid objectid")
        return ObjectId(v)

class VideoMeta(BaseModel):
    id: Optional[PyObjectId] = Field(alias="_id")
    title: str
    description: Optional[str] = None
    source: Optional[str] = None
    type: str  # open_source | external | user
    video_url: str
    embed_html: Optional[str] = None
    thumbnail_url: Optional[str] = None
    duration: Optional[int] = None
    license: Optional[str] = None
    author_name: Optional[str] = None
    author_avatar: Optional[str] = None
    created_at: datetime
    fetched_at: Optional[datetime] = None

    class Config:
        arbitrary_types_allowed = True
        json_encoders = {ObjectId: str}

"""
Implementation of many sources: mix of official APIs and HTML scrapers / oEmbed.
We include ~15 sources. Each fetch function returns list of metadata dicts.
"""

import requests
from app.config import PEXELS_API_KEY, PIXABAY_API_KEY, USER_AGENT, BACKEND_HOST
from datetime import datetime
from app.services.video_service import cache_get, cache_set
from app.services.avatars import author_name_for_source, avatar_url_for_source

HEADERS = {"User-Agent": USER_AGENT}

def _norm_item(video_url, title, thumb=None, source="unknown", duration=None, license="unknown"):
    return {
        "title": title or "Vidéo",
        "description": None,
        "source": source,
        "type": "open_source",
        "video_url": video_url,
        "thumbnail_url": thumb,
        "duration": duration,
        "license": license,
        "author_name": author_name_for_source(source),
        "author_avatar": avatar_url_for_source(source),
        "created_at": datetime.utcnow(),
        "fetched_at": datetime.utcnow()
    }

def fetch_pexels(query="africa", per_page=15):
    if not PEXELS_API_KEY: return []
    headers = {"Authorization": PEXELS_API_KEY}
    params = {"query": query, "per_page": per_page}
    try:
        r = requests.get("https://api.pexels.com/videos/search", headers=headers, params=params, timeout=10)
        if r.status_code != 200: return []
        data = r.json()
        items = []
        for v in data.get("videos", []):
            files = v.get("video_files", [])
            video_url = files[0].get("link") if files else None
            thumb = v.get("image")
            title = v.get("user", {}).get("name") or v.get("url") or "Pexels video"
            if video_url:
                items.append(_norm_item(video_url, title, thumb, "Pexels", v.get("duration"), "pexels"))
        return items
    except:
        return []

def fetch_pixabay(query="africa", per_page=15):
    if not PIXABAY_API_KEY: return []
    params = {"key": PIXABAY_API_KEY, "q": query, "per_page": per_page}
    try:
        r = requests.get("https://pixabay.com/api/videos/", params=params, timeout=10)
        if r.status_code != 200: return []
        data = r.json()
        items = []
        for v in data.get("hits", []):
            video_url = v.get("videos", {}).get("medium", {}).get("url")
            thumb = v.get("pictures", [{}])[0].get("picture")
            title = v.get("tags") or "Pixabay video"
            if video_url:
                items.append(_norm_item(video_url, title, thumb, "Pixabay", v.get("duration"), "pixabay"))
        return items
    except:
        return []

def fetch_mixkit(per_page=12):
    items = []
    try:
        r = requests.get("https://mixkit.co/free-stock-video", headers=HEADERS, timeout=10)
        if r.status_code == 200:
            from bs4 import BeautifulSoup
            soup = BeautifulSoup(r.text, "html.parser")
            cards = soup.select("a.js-video-link")[:per_page]
            for c in cards:
                href = c.get("href")
                title = c.get("title") or "Mixkit video"
                if href:
                    page = requests.get("https://mixkit.co"+href, headers=HEADERS, timeout=8)
                    if page.status_code==200:
                        sp = BeautifulSoup(page.text,"html.parser")
                        vtag = sp.find("a", {"class":"download-button"})
                        video_url = vtag.get("href") if vtag else None
                        thumb = sp.find("meta", {"property":"og:image"})
                        thumb = thumb.get("content") if thumb else None
                        if video_url:
                            items.append(_norm_item(video_url, title, thumb, "Mixkit", None, "mixkit"))
    except Exception:
        pass
    return items

def fetch_generic_site_for_videos(url, source_name, per_page=10):
    items = []
    try:
        r = requests.get(url, headers=HEADERS, timeout=10)
        if r.status_code != 200: return items
        from bs4 import BeautifulSoup
        soup = BeautifulSoup(r.text, "html.parser")
        videos = soup.find_all("video")[:per_page]
        for v in videos:
            src = v.get("src") or (v.find("source") and v.find("source").get("src"))
            if not src: continue
            thumb = None
            img = v.get("poster")
            if img: thumb = img
            title = v.get("title") or f"{source_name} clip"
            items.append(_norm_item(src, title, thumb, source_name, None, "public"))
    except Exception:
        pass
    return items

def fetch_all_sources():
    results = []
    results += fetch_pexels()
    results += fetch_pixabay()
    results += fetch_mixkit()
    generic_sites = [
        ("https://coverr.co/", "Coverr"),
        ("https://www.lifeofvids.com/", "LifeOfVids"),
        ("https://www.videvo.net/", "Videvo"),
        ("https://www.vidsplay.com/", "Vidsplay"),
        ("https://www.splitshire.com/", "Splitshire"),
        ("https://motionplaces.com/", "MotionPlaces"),
        ("https://mazwai.com/", "Mazwai"),
        ("https://archive.org/details/movies", "ArchiveOrg"),
        ("https://commons.wikimedia.org/wiki/Category:Videos", "Wikimedia"),
        ("https://distillvideo.com/", "Distill"),
        ("https://clipstill.com/", "Clipstill"),
    ]
    for url, name in generic_sites:
        results += fetch_generic_site_for_videos(url, name, per_page=8)
    # dedupe by video_url
    seen = set()
    uniq = []
    for r in results:
        if r["video_url"] and r["video_url"] not in seen:
            seen.add(r["video_url"])
            uniq.append(r)
    return uniq

# simple in-memory cache used for API calls (also saved to Mongo optional)
import time
_cache = {}

def requests_cache_get(key):
    v = _cache.get(key)
    if not v: return None
    payload, expires = v
    if expires < time.time():
        del _cache[key]
        return None
    return payload

def requests_cache_set(key, payload, ttl=3600):
    _cache[key] = (payload, time.time() + ttl)

'use client';
import { useEffect, useRef, useState } from "react";
import VideoCard from "./VideoCard";

export default function VideoFeed(){
  const [videos, setVideos] = useState([]);
  const [page, setPage] = useState(1);
  const containerRef = useRef(null);
  const activeIndexRef = useRef(0);
  const loadingRef = useRef(false);

  useEffect(()=>{ loadPage(1); }, []);

  const loadPage = async (p) => {
    if (loadingRef.current) return;
    loadingRef.current = true;
    try {
      const res = await fetch(`/api/videos?page=${p}&limit=10`);
      const data = await res.json();
      if (data && data.length) {
        setVideos(v => p === 1 ? data : [...v, ...data]);
        setPage(p);
      }
    } catch(e){ console.error(e) }
    finally { loadingRef.current = false; }
  };

  const pauseAllExcept = (index) => {
    const vids = document.querySelectorAll(".vh-video");
    vids.forEach((v,i) => {
      try { if (i === index) { v.play(); } else { v.pause(); } } catch(e){}
    });
  };

  useEffect(() => {
    const container = containerRef.current;
    if (!container) return;
    const onScroll = () => {
      const scrollPos = container.scrollTop;
      const height = container.clientHeight;
      const newIndex = Math.round(scrollPos / height);
      if (newIndex !== activeIndexRef.current) {
        activeIndexRef.current = newIndex;
        pauseAllExcept(newIndex);
        if (newIndex >= videos.length - 3) {
          loadPage(page + 1);
        }
      }
    };
    container.addEventListener("scroll", onScroll);
    return () => container.removeEventListener("scroll", onScroll);
  }, [videos, page]);

  useEffect(()=> {
    if (videos.length>0) setTimeout(()=> pauseAllExcept(0), 200);
  }, [videos]);

  return (
    <div ref={containerRef} className="h-screen overflow-y-scroll snap-y snap-mandatory">
      {videos.map((v,i)=>( 
        <div key={v._id} className="h-screen w-full snap-start flex items-center justify-center bg-black">
          <VideoCard video={v} index={i} />
        </div>
      ))}
      <div style={{height:60}} />
    </div>
  );
}

===== FILE: NOTES_SCRAPER_POLICIES.md =====

Important legal & practical notes:
- Only scrape content that is explicitly allowed. Use official APIs where possible.
- Respect robots.txt, rate limits, and the platform's Terms of Service.
- For websites that require JS rendering, consider Playwright for debug only (avoid mass scraping).
- Cache results; don't reload the same pages too often (CONFIG: CACHE_TTL_SECONDS).
- Attribution: if a source requires attribution (CC-BY), show it in the UI.
- For heavy use, move heavy fetches to worker processes (Celery / RQ) and use S3/CDN if you store assets.
